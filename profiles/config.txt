# Boot configuration for ClockworkPi uConsole with Raspberry Pi CM5.
#
# Display settings here were reconciled against ak-rex's known-working Debian
# image after an earlier build of this project produced a dark panel.

arm_64bit=1
disable_overscan=1

[cm5]
# Kernel. The 4K-page package deliberately installs into this same boot slot,
# so the filename says cm5-git even though it is the 4k variant.
kernel=vmlinuz-linux-uconsole-cm5-git
initramfs initramfs-linux-uconsole-cm5-git.img followkernel

# uConsole hardware: DSI panel (cw,cwu50), OCP8178 backlight, AXP228 PMIC
dtoverlay=clockworkpi-uconsole-cm5

# Pi 5 / CM5 KMS display driver
dtoverlay=vc4-kms-v3d-pi5,cma-384

# PCIe enabled to match the working reference configuration. The mini-PCIe
# slot carries the 4G module; gen=3 mirrors ak-rex's image.
dtparam=pciex1
dtparam=pciex1_gen=3

# USB port current limit increase for Pi 5 / CM5
usb_max_current_enable=1

# Audio
dtparam=audio=on
dtoverlay=audremap-pi5,pins_12_13

# ramoops: keep the tail of the kernel log in a RAM region that a warm reset does
# not clear, so it can be read from /sys/fs/pstore on the next boot.
#
# This exists because of a specific, repeated failure: every hard hang on this
# machine destroys the journal, so the one thing we most need -- what the kernel
# was doing in the instant before it stopped -- has never once been recoverable.
# Trying to read it from the console does not work either: cmdline.txt sets
# console=tty1, which is where sway runs, so switching to tty2 to watch a suspend
# means the printk output goes to the terminal you just left.
#
# With the watchdog now rebooting a wedged machine automatically, a RAM console is
# what turns that reboot into evidence rather than just a recovery.
# console-size is the load-bearing parameter, and it defaults to ZERO.
#
# ramoops' record-size buffer only receives an oops or panic dump -- kmsg_dump()
# has to actually run to write it. A HANG produces neither, and neither does a
# watchdog reset: the CPU is simply restarted, so no code runs to save anything.
# Set up with record-size alone, /sys/fs/pstore comes back empty after exactly
# the failure we are trying to diagnose. It did.
#
# console-size creates a second ring that receives the kernel console
# CONTINUOUSLY, so whatever printk emitted right up to the instant the machine
# stopped is still in RAM after a warm reset -- including pm_print_times naming
# the last device to suspend.
#
# The reserved region is 64 KiB total (base 0xb000000). 32 KiB of console plus
# the 16 KiB record buffer fits with room to spare.
dtoverlay=ramoops-pi5,console-size=0x8000

[all]
# Stop the firmware bringing up its own default DSI display so that the
# clockworkpi overlay owns the panel. Omitting this leaves the screen dark.
ignore_lcd=1
max_framebuffers=2

# USB host mode for the uConsole internal/external USB path
dtoverlay=dwc2,dr_mode=host

# External antenna on the Raspberry Pi CM U.FL connector
dtparam=ant2

# GPIO 10 senses headphone insertion; GPIO 11 enables the speaker amplifier.
gpio=10=ip,np
gpio=11=op,dh
